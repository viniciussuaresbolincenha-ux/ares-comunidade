# vendas.py # ------------------------------------------------- # Sistema de vendas simples (linha de comando) # ------------------------------------------------- # Autor: Zenato # Data : 05/11/2026 # ------------------------------------------------- from dataclasses import dataclass, field from typing import List, Dict @dataclass class Produto: """Representa um produto no estoque.""" codigo: int nome: str preco: float estoque: int def __str__(self): return f"[{self.codigo}] {self.nome} - R${self.preco:.2f} (Estoque: {self.estoque})" @dataclass class ItemCarrinho: """Um item que o cliente adicionou ao carrinho.""" produto: Produto quantidade: int @property def subtotal(self) -> float: return self.produto.preco * self.quantidade class SistemaVendas: def __init__(self): self.produtos: Dict[int, Produto] = {} self.carrinho: List[ItemCarrinho] = [] self._proximo_codigo = 1 # ------------------- Gestão de Produtos ------------------- def cadastrar_produto(self, nome: str, preco: float, estoque: int): codigo = self._proximo_codigo self._proximo_codigo += 1 self.produtos[codigo] = Produto(codigo, nome, preco, estoque) print(f"Produto cadastrado: {self.produtos[codigo]}") def listar_produtos(self): if not self.produtos: print("Nenhum produto cadastrado.") return print("\n=== Lista de Produtos ===") for p in self.produtos.values(): print(p) # ------------------- Carrinho ------------------- def adicionar_ao_carrinho(self, codigo: int, quantidade: int): if codigo not in self.produtos: print("Código de produto inexistente.") return produto = self.produtos[codigo] if quantidade <= 0: print("Quantidade deve ser maior que zero.") return if produto.estoque < quantidade: print(f"Estoque insuficiente. Disponível: {produto.estoque}") return # Verifica se já há o mesmo produto no carrinho for item in self.carrinho: if item.produto.codigo == codigo: # Atualiza quantidade if produto.estoque < item.quantidade + quantidade: print(f"Estoque insuficiente para somar ao carrinho. Disponível: {produto.estoque - item.quantidade}") return item.quantidade += quantidade print(f"Adicionado ao carrinho: {produto.nome} (Qtde: {item.quantidade})") return # Caso ainda não exista no carrinho self.carrinho.append(ItemCarrinho(produto, quantidade)) print(f"Adicionado ao carrinho: {produto.nome} (Qtde: {quantidade})") def mostrar_carrinho(self): if not self.carrinho: print("Carrinho vazio.") return print("\n=== Carrinho ===") total = 0.0 for idx, item in enumerate(self.carrinho, start=1): print(f"{idx}. {item.produto.nome} - R${item.produto.preco:.2f} x {item.quantidade} = R${item.subtotal:.2f}") total += item.subtotal print(f"Total: R${total:.2f}") # ------------------- Finalizar Compra ------------------- def finalizar_compra(self): if not self.carrinho: print("Carrinho vazio. Nada a finalizar.") return self.mostrar_carrinho() confirma = input("\nDeseja confirmar a compra? (s/n): ").strip().lower() if confirma != "s": print("Compra cancelada.") return # Atualiza estoque for item in self.carrinho: item.produto.estoque -= item.quantidade # Limpa o carrinho self.carrinho.clear() print("Compra concluída! Obrigado e volte sempre.") # ------------------- Menu Interativo ------------------- def exibir_menu(self): while True: print("\n=== Sistema de Vendas ===") print("1 - Cadastrar produto") print("2 - Listar produtos") print("3 - Adicionar ao carrinho") print("4 - Mostrar carrinho") print("5 - Finalizar compra") print("0 - Sair") escolha = input("Escolha uma opção: ").strip() if escolha == "1": nome = input("Nome do produto: ").strip() preco = float(input("Preço (R$): ").replace(",", ".")) estoque = int(input("Quantidade em estoque: ")) self.cadastrar_produto(nome, preco, estoque) elif escolha == "2": self.listar_produtos() elif escolha == "3": codigo = int(input("Código do produto: ")) quantidade = int(input("Quantidade: ")) self.adicionar_ao_carrinho(codigo, quantidade) elif escolha == "4": self.mostrar_carrinho() elif escolha == "5": self.finalizar_compra() elif escolha == "0": print("Encerrando o sistema. Até a próxima!") break else: print("Opção inválida. Tente novamente.") if __name__ == "__main__": sistema = SistemaVendas() sistema.exibir_menu()
